# sampled per liner). Build agent: C1_liners. Deviations from the card: ../DEVIATIONS.md; real products: ../ASSETS.md.
# Robot: the repo's two-Panda B4 robot (portable_demo.env.scene(seed, 'test') with every non-robot body stripped), stock Franka
# hands, top-down grasps only (the standard agent API). Actions are data.ctrl rows (nu = 18): 14 residual arm torques (step()
# adds the arm gravity/bias compensation, so a zero row floats the arms) and 4 finger opening targets (0 closed .. .045 open).
# Scene (every scene, exploration and deployment, has the same layout): six look-alike non-slip liners (rigid 70 x 150 x 6 mm
# plates, compliant contact) lying on the table in two rows, a fixed 20 deg rack (sloped board on a stand), two identical short glass tumblers, a shallow
# water pan and a flat practice tray.
# Physics: the cup-to-liner friction of every (cup, liner) pair is an explicit <contact><pair> (the hidden value); liners grip
# the rack and table with a fixed rubber friction of 1.0; a bare glass on the varnished rack (mu .30 < tan 20 deg) slides.
# Settings from probes batch1 #5 (creep): timestep 1 ms, friction solref .001 (clamped to 2 dt), noslip 20, multiccd.
# Rule `wet` (reads only sim state): a liner whose centre is inside the pan's water volume becomes wet once: its cup pair
# friction switches to its wet value and it turns glossy (appearance). No teleports, no object actuators.
# Hidden cells: mu_dry (P, 6 dry cup-liner friction coefficients), mu_wet (P, 6 wet values).
# Damage `cup_fell`: a released glass that was resting on the rack ends below the rack's low edge or tips past 60 deg.

def _u(r, lo_hi): return float(r.uniform(*lo_hi))


def sample_liners(r, n_hold):
    # Card distribution (not per-seed screening): n_hold liners hold dry (U .46-1.0), the rest slip (U .15-.27), shuffled.
    # Wet: mu_wet = mu x [0.3, 1.2], cap 1.0, never in the gray zone, sampled per liner (DECISIONS C1). Stratified so that 1 or
    # 2 liners (never all holders) hold when wet; whether the dry-best liner still holds wet is left to chance.
    hold = np.array([True] * n_hold + [False] * (N_LINER - n_hold)); r.shuffle(hold)
    mu = [_u(r, MU_HOLD) if h else _u(r, MU_SLIP) for h in hold]
    holders = [i for i in range(N_LINER) if hold[i]]
    n_wet = int(r.integers(1, min(2, n_hold - 1) + 1))
    wet_hold = set(r.choice(holders, n_wet, replace=False).tolist())
    mu_wet = []
    for i in range(N_LINER):
        if i in wet_hold: mu_wet.append(_u(r, (max(MU_HOLD[0], .3 * mu[i]), min(1., 1.2 * mu[i]))))
        else: mu_wet.append(_u(r, (.3 * mu[i], min(GRAY[0], 1.2 * mu[i]))))
    clip = lambda x: min(round(x, 4), .3099) if x < GRAY[0] else round(x, 4)    # rounding never lands on the gray-zone edge
    return [clip(x) for x in mu], [clip(x) for x in mu_wet]


def parameters(seed):
    r = np.random.default_rng(np.random.SeedSequence([seed, 7101]))
    mu, mu_wet = sample_liners(r, int(r.integers(2, 4)))
    return dict(mu_dry=mu, mu_wet=mu_wet)


def transfer_parameters(params, seed):
    r = np.random.default_rng(np.random.SeedSequence([seed, 7102]))
    mu, mu_wet = sample_liners(r, int(r.integers(2, 4)))
    return dict(mu_dry=mu, mu_wet=mu_wet)


def _cup_I():
    # thin-walled glass tumbler with a thick base (estimate): inertia of a solid cylinder of the same size, scaled by 0.8
    m, r, h = CUP_MASS, CUP_R, 2 * CUP_HH
    return _f(.8 * np.array([m * (3 * r * r + h * h) / 12] * 2 + [m * r * r / 2]))


def _f(v): return ' '.join(f'{x:.6g}' for x in np.atleast_1d(v))

def ids(m):
    arms = [[m.joint(f'{s}_joint{i}').id for i in range(1, 8)] for s in ('left', 'right')]
    owner = np.zeros(m.ngeom, int)          # 0 other, 1 robot, 10+c cup, 20+i liner, 30 rack, 31 table/floor/pan/tray
    for g in range(m.ngeom):
        bn = m.body(m.geom_bodyid[g]).name; gn = m.geom(g).name
        if bn.startswith(('left_', 'right_')): owner[g] = 1
        elif bn.startswith('cup_'): owner[g] = 10 + int(bn[4:]) - 1
        elif bn.startswith('liner_'): owner[g] = 20 + int(bn[6:]) - 1 if gn.endswith('_g') else 0
        elif gn == 'rack_board': owner[g] = 30
        elif gn.startswith(('table', 'floor', 'pan_', 'tray', 'rack_leg')): owner[g] = 31
    return dict(q=[m.jnt_qposadr[a].tolist() for a in arms], v=[m.jnt_dofadr[a].tolist() for a in arms],
                fingers=[int(m.jnt_qposadr[m.joint(f'{s}_finger_{f}').id]) for s in ('left', 'right') for f in 'ab'],
                owner=owner.tolist(), cups=[int(m.body(f'cup_{c + 1}').id) for c in range(N_CUP)],
                liners=[int(m.body(f'liner_{i + 1}').id) for i in range(N_LINER)],
                liner_g=[int(m.geom(f'liner_{i + 1}_g').id) for i in range(N_LINER)],
                pairs=[[int(m.pair(f'p_{c + 1}_{i + 1}').id) for i in range(N_LINER)] for c in range(N_CUP)],
                wet_mat=int(m.material('liner_wet').id), tips=[int(m.site('left_tip').id), int(m.site('right_tip').id)],
                robot=[g for g in range(m.ngeom) if owner[g] == 1])


def init_state(seed, target, params):
    p = params or parameters(seed); t = TGT.get(target) if target is not None else None
    return dict(mu_dry=list(p['mu_dry']), mu_wet=list(p['mu_wet']), wet=[False] * N_LINER, wet_fired=False,
                ok_t=[0.] * N_CUP, ref=[None] * N_CUP, void=[False] * N_CUP, hold=[0.] * N_CUP, on_liner=[-1] * N_CUP, was_on_rack=[False] * N_CUP,
                release_z=[None] * N_CUP, fell_reason=None,
                cup_fell=False, need_cups=t['cups'] if t else None, need_wet=bool(t['wet']) if t else False)


def contacts(m, d, k):
    # Per-tick contact summary from sim state: which owner codes touch which.
    own = k['owner']; touch = set()
    for i in range(d.ncon):
        a, b = own[d.contact.geom1[i]], own[d.contact.geom2[i]]
        if a and b: touch.add((a, b)); touch.add((b, a))
    return touch


def robot_supported_objects(touch, include_cups=True):
    """Mobile objects connected to either robot by contact, ignoring statics.

    A spare liner can transmit robot support to a glass on another liner.
    Static rack/table contacts must not join every resting object to a robot.
    """
    mobile={1,*range(20,20+N_LINER)}
    if include_cups:mobile.update(range(10,10+N_CUP))
    edges=[(a,b) for a,b in touch if a in mobile and b in mobile]
    supported={1};pending=[1]
    while pending:
        node=pending.pop()
        for a,b in edges:
            if a==node and b not in supported:supported.add(b);pending.append(b)
            elif b==node and a not in supported:supported.add(a);pending.append(a)
    return supported


def step(m, d, state):
    k = state.get('ids') or state.setdefault('ids', ids(m))
    dofs = k['v'][0] + k['v'][1]
    d.qfrc_applied[dofs] = d.qfrc_bias[dofs]
    # rule `wet`: a liner whose centre is in the pan's water volume gets wet once (pair friction + appearance)
    for i, b in enumerate(k['liners']):
        if state['wet'][i]: continue
        x = d.xpos[b]
        if abs(x[0] - PAN_C[0]) < PAN_IN[0] and abs(x[1] - PAN_C[1]) < PAN_IN[1] and x[2] < WATER_TOP:
            state['wet'][i] = True; state['wet_fired'] = True
            for c in range(N_CUP): m.pair_friction[k['pairs'][c][i], :2] = state['mu_wet'][i]
            m.geom_matid[k['liner_g'][i]] = k['wet_mat']
    mujoco.mj_step(m, d)
    T = contacts(m, d, k); dt = m.opt.timestep; supported = robot_supported_objects(T); liner_supported = robot_supported_objects(T, include_cups=False)
    for c, b in enumerate(k['cups']):
        cc = 10 + c
        held = cc in supported
        lin = [i for i in range(N_LINER) if (cc, 20 + i) in T]
        on_rack_liner = [i for i in lin if (20 + i, 30) in T and 20 + i not in liner_supported]
        # count 'was on the rack' only while released: a glass carried back to the table by the robot is not a fall
        if held: state['was_on_rack'][c] = False; state['release_z'][c] = float(d.xpos[b, 2]); state['void'][c] = False
        elif (cc, 30) in T or on_rack_liner: state['was_on_rack'][c] = True
        up = float(d.xmat[b].reshape(3, 3)[:, 2] @ N_RACK) > np.cos(np.radians(UPRIGHT_DEG))
        ok = not held and len(on_rack_liner) == 1 and (cc, 30) not in T and (cc, 31) not in T and up
        if ok and state['ok_t'][c] > 0 and state['on_liner'][c] != on_rack_liner[0]: ok = False
        # the settled reference is fixed once, 0.5 s after the last robot contact; a placement that loses it (drifts > 5 mm or
        # stops qualifying) stays void until the robot touches the glass again (no later hands-off re-settling window)
        if state['void'][c]: ok = False
        if ok:
            state['ok_t'][c] += dt; state['on_liner'][c] = on_rack_liner[0]
            if state['ref'][c] is None and state['ok_t'][c] >= SETTLE_S: state['ref'][c] = d.xpos[b].tolist()
            if state['ref'][c] is not None:
                if np.linalg.norm(d.xpos[b] - np.array(state['ref'][c])) > MOVE_TOL: ok = False
                else: state['hold'][c] = state['ok_t'][c] - SETTLE_S
        if not ok:
            if state['ref'][c] is not None and not held: state['void'][c] = True
            state['ok_t'][c] = 0.; state['ref'][c] = None; state['hold'][c] = 0.; state['on_liner'][c] = -1
        # damage: a released glass that was on the rack ends up below the rack's low edge, or tips over
        if state['was_on_rack'][c] and not held:
            tilt = float(d.xmat[b, 8])
            if d.xpos[b, 2] < RACK_LOW_Z - .01 or tilt < np.cos(np.radians(60)):
                state['cup_fell'] = True; state['fell_reason'] = state['fell_reason'] or 'slid_off_or_tipped_from_rack'
        # extension: tip-over anywhere, or an unheld free drop from the last robot-contact height onto table/pan/tray/floor
        if not held:
            rz = state['release_z'][c]
            if float(d.xmat[b, 8]) < np.cos(np.radians(TIP_DEG)):
                state['cup_fell'] = True; state['fell_reason'] = state['fell_reason'] or 'tipped_over_unheld'
            elif rz is not None and rz - float(d.xpos[b, 2]) > DROP_TOL and (cc, 31) in T:
                state['cup_fell'] = True; state['fell_reason'] = state['fell_reason'] or 'dropped_unheld'


def success(model, data, state, target):
    if state['cup_fell']: return False
    good = [c for c in range(N_CUP) if state['hold'][c] >= HOLD_S and (not state['need_wet'] or state['wet'][state['on_liner'][c]])]
    n = state['need_cups'] or 1
    return len({state['on_liner'][c] for c in good}) >= n


def damage(model, data, state): return dict(cup_fell=state['cup_fell'])


# ---------------------------------------------------------------- externally measurable partial credit (final state)
# Per required glass (1, or 2 for C1-T3), each on a different liner, evaluated at the end of the episode:
#   0.25  a liner lies on the rack board (touching the board, not the table, no robot contact, within 15 deg of flat on the
#         board); C1-T2: that liner must have been wetted in the pan;
#   0.50  plus a released glass stands on that liner, upright within 15 deg of the board normal, touching neither bare board nor
#         table, no robot contact on glass or liner;
#   0.75  plus it has stayed within 5 mm of its settled position (settle 0.5 s after last robot contact) for at least 10 s
#         (a placement that drifted > 5 mm after settling is void until renewed robot contact: no hands-off re-settling);
#   1.00  plus for at least 20 s (= source success).
# Score = mean over the required glasses. Global veto: any fallen glass (cup_fell) scores 0.0.
SCORE_STAGES = ((.25, 'liner_on_rack'), (.50, 'glass_standing_on_liner'), (.75, 'held_10s'), (1.0, 'held_20s'))


def score_detail(model, data, state, target):
    k = state.get('ids') or state.setdefault('ids', ids(model))
    T = contacts(model, data, k); supported = robot_supported_objects(T, include_cups=False); n = state['need_cups'] or 1; need_wet = state['need_wet']
    flat = lambda i: float(data.xmat[k['liners'][i]].reshape(3, 3)[:, 2] @ N_RACK) > np.cos(np.radians(UPRIGHT_DEG))
    liners = [i for i in range(N_LINER) if (20 + i, 30) in T and (20 + i, 31) not in T and 20 + i not in supported and flat(i)
              and (not need_wet or state['wet'][i])]
    items = []
    for c in range(N_CUP):
        i = state['on_liner'][c]
        if state['ok_t'][c] > 0 and i in liners:
            h = state['hold'][c]
            items.append((1.0 if h >= HOLD_S else .75 if h >= 10. else .5, i, c, h))
    items.sort(reverse=True); used, per = set(), []
    for v, i, c, h in items:
        if i not in used and len(per) < n: used.add(i); per.append(dict(value=v, liner=i + 1, glass=c + 1, hold_s=round(h, 3)))
    for i in liners:
        if len(per) >= n: break
        if i not in used: used.add(i); per.append(dict(value=.25, liner=i + 1, glass=None, hold_s=0.))
    raw = sum(p['value'] for p in per) / n
    return dict(score=0. if state['cup_fell'] else float(raw), raw_before_veto=float(raw), items=per, required=n,
                qualifying_liners=[i + 1 for i in liners], cup_fell=bool(state['cup_fell']), fell_reason=state.get('fell_reason'))


def score(model, data, state, target): return score_detail(model, data, state, target)['score']

def wrist_wrench(model, data):
    # No flex in this scene: native wrist sensors see every contact.
    out = []
    for s in ('left', 'right'):
        R = data.site(s + '_wrist').xmat.reshape(3, 3)
        out += [R @ data.sensor(s + '_force').data, R @ data.sensor(s + '_torque').data]
    return np.concatenate(out)

# ---------------------------------------------------------------- privileged reference controller (top-down grasps only)
def r_down(yaw): return Rotation.from_euler('z', yaw).as_matrix() @ DOWN


def ik(m, scratch, k, arm, target, q0, R, iters=15):
    q, v, tip = k['q'][arm], k['v'][arm], k['tips'][arm]
    scratch.qpos[q] = q0
    jp, jr = np.zeros((3, m.nv)), np.zeros((3, m.nv)); lo, hi = m.jnt_range[[m.dof_jntid[i] for i in v]].T
    for _ in range(iters):
        mujoco.mj_kinematics(m, scratch); mujoco.mj_comPos(m, scratch)
        err = np.r_[target - scratch.site_xpos[tip], .5 * Rotation.from_matrix(R @ scratch.site_xmat[tip].reshape(3, 3).T).as_rotvec()]
        if np.linalg.norm(err) < 1e-5: break
        mujoco.mj_jacSite(m, scratch, jp, jr, tip)
        J = np.vstack([jp[:, v], .5 * jr[:, v]])
        scratch.qpos[q] = np.clip(scratch.qpos[q] + np.clip(J.T @ np.linalg.solve(J @ J.T + 1e-4 * np.eye(6), err), -.08, .08), lo + .01, hi - .01)
    return scratch.qpos[q].copy()


YAW = [0., np.pi]     # left: jaw axis along world X; right: jaw axis along world X (base turned 180 deg)


class Driver:
    # Scripted Cartesian plan per arm tracked by joint PD; records ctrl rows. Only reference() chooses liners with hidden values.
    def __init__(self, seed, target, params):
        self.m, self.d = build(seed, target, params); self.st = init_state(seed, target, params); self.k = ids(self.m)
        self.scratch = mujoco.MjData(self.m); self.acts = []
        self.pos = [self.d.site_xpos[t].copy() for t in self.k['tips']]
        self.qt = [self.d.qpos[self.k['q'][a]].copy() for a in (0, 1)]
        self.jaw = [.045, .045]
        self.R = [r_down(YAW[0]), r_down(YAW[1])]
        self.hook = None

    def tick(self):
        m, d, k = self.m, self.d, self.k; u = np.zeros(m.nu)
        for arm in (0, 1):
            u[7 * arm:7 * arm + 7] = np.clip(KP * (self.qt[arm] - d.qpos[k['q'][arm]]) - KD * d.qvel[k['v'][arm]], -LIM, LIM)
        u[14:16] = self.jaw[0]; u[16:18] = self.jaw[1]
        d.ctrl[:] = u; self.acts.append(u.copy()); step(m, d, self.st)
        if self.hook: self.hook(self)

    def move(self, arm, goal=None, jaw=None, n=600, both=None, yaw=None, linear=False, every=10):
        # Smoothstep Cartesian move of one arm (optionally a second arm simultaneously: both=(arm2, goal2, jaw2)); the tool
        # stays pointing down, and its yaw turns smoothly to `yaw` if given.
        plans = [(arm, goal, jaw)] + ([both] if both else [])
        st = []
        for a, g, j in plans:
            p0 = self.pos[a].copy(); g = p0 if g is None else np.asarray(g, float)
            R1 = self.R[a] if (yaw is None or a != arm) else r_down(yaw)
            sl = Slerp([0., 1.], Rotation.from_matrix([self.R[a], R1]))
            st.append((a, p0, g, self.jaw[a], self.jaw[a] if j is None else j, sl, R1))
        for i in range(n):
            f = min(1., (i + 1) / max(1, n - 60)); f = (i + 1) / n if linear else f * f * (3 - 2 * f)
            for a, p0, g, j0, j1, sl, R1 in st:
                if i % every == 0 or i == n - 1:
                    self.qt[a] = ik(self.m, self.scratch, self.k, a, p0 + (g - p0) * f, self.qt[a], sl(f).as_matrix())
                self.jaw[a] = j0 + (j1 - j0) * f
            self.tick()
        for a, p0, g, j0, j1, sl, R1 in st: self.pos[a] = g.copy(); self.R[a] = R1

    def wait(self, n):
        for _ in range(n): self.tick()

    def jmove(self, arm, goal, n=1200):
        # joint-space smoothstep from the current joint target to the IK solution at goal (used for the first transfer)
        q0 = self.qt[arm].copy(); q1 = q0.copy()
        for _ in range(25): q1 = ik(self.m, self.scratch, self.k, arm, np.asarray(goal, float), q1, self.R[arm])
        for i in range(n):
            f = min(1., (i + 1) / max(1, n - 60)); f = f * f * (3 - 2 * f)
            self.qt[arm] = q0 + (q1 - q0) * f; self.tick()
        self.pos[arm] = np.asarray(goal, float).copy()


# reference geometry
PAD_BELOW_TIP = .025                    # finger pads extend 25 mm below the tool point when pointing down
LINER_GRASP_Z = TABLE_Z + .002 + PAD_BELOW_TIP     # pad bottoms 2 mm above the table
CUP_GRASP_DZ = .040                     # tool point this far above the glass centre (pads cover the upper glass)
SAFE_Z = 1.06
PARK = [np.array([-.12, -.10, 1.10]), np.array([.28, -.18, 1.10])]
LINER_CLEAR, CUP_CLEAR = .003, .0015
# Liner pinch: a gentle grip (finger target 20 mm, jaw gap 40 mm on the 70 mm liner: about 18 N per finger). A full squeeze
# (target 0, 40 N) issued as one jaw jump, as the stock agent tool does, makes the pads ride over the 6 mm compliant edge and
# lose the liner (#97 stock-controller replay, polish 2026-10-29; DEVIATIONS 17).
LINER_JAW = .020
CUP_UPSLOPE = 0.     # glass stood at the liner centre (holders hold from centre or +-25 mm, see DEVIATIONS 3)


def slot_point(slot, along=0.):
    # top-surface point of the rack board at slot x, `along` m up the slope from the board centre line
    p = RACK_TOP_C + U_SLOPE * along; p = p.copy(); p[0] = SLOTS_X[slot]; return p


def place_liner(D, i, slot, wet=False):
    # Left arm: pinch liner i by its long edges from above, (dip it in the pan), lay it on the rack slot, release.
    L = 0; x, y = LINER_XY[i]
    D.move(L, [x, y, SAFE_Z], .045, 900)
    D.move(L, [x, y, LINER_GRASP_Z + .03], .045, 700)
    D.move(L, [x, y, LINER_GRASP_Z], .045, 700)
    D.move(L, None, LINER_JAW, 350)
    D.move(L, [x, y, SAFE_Z], LINER_JAW, 1400)
    if wet:
        D.move(L, [PAN_C[0], PAN_C[1], SAFE_Z], LINER_JAW, 900)
        D.move(L, [PAN_C[0], PAN_C[1], TABLE_Z + PAN_FLOOR + .007 + PAD_BELOW_TIP], LINER_JAW, 900)
        D.wait(200)
        D.move(L, [PAN_C[0], PAN_C[1], SAFE_Z], LINER_JAW, 900)
    s = slot_point(slot)
    # hold the liner horizontal; its up-slope end touches the board first; lower until ~10 mm of drop remains
    bottom = s[2] + LINER_H[1] * np.tan(ANGLE) + LINER_CLEAR      # liner underside; its up-slope end clears the board
    tip = np.array([s[0], s[1], bottom + .002 + PAD_BELOW_TIP])
    D.move(L, [-.06, -.05, SAFE_Z], LINER_JAW, 800)        # waypoint: keep clear of the parked right arm
    D.move(L, [s[0], s[1], SAFE_Z], LINER_JAW, 1000)
    D.move(L, tip, LINER_JAW, 900)
    D.move(L, None, .045, 300)
    D.wait(300)
    D.move(L, tip + [0, -.03, .10], .045, 700)
    D.move(L, PARK[L], .045, 1000)


def place_cup(D, c, slot):
    # Right arm: grasp glass c, carry it above the liner centre on the slot, lower until the up-slope rim is ~3 mm above the
    # liner, release, retreat.
    R = 1; x, y = CUP_XY[c]; cz = TABLE_Z + CUP_HH
    D.move(R, [x, y, SAFE_Z], .045, 900)
    D.move(R, [x, y, cz + CUP_GRASP_DZ], .045, 700)
    D.move(R, None, .0, 350)
    D.move(R, [x, y, SAFE_Z], 0., 900)
    s = slot_point(slot, CUP_UPSLOPE) + N_RACK * 2 * LINER_H[2]   # liner top centre
    # glass upright: its lowest point over the slope is the up-slope rim at +y
    centre = np.array([s[0], s[1], s[2] + CUP_R * np.tan(ANGLE) + CUP_CLEAR + CUP_HH])
    D.move(R, [centre[0], centre[1], SAFE_Z], 0., 1000)
    D.move(R, centre + [0, 0, CUP_GRASP_DZ], 0., 1000)
    D.move(R, None, .045, 600)
    D.move(R, centre + [0, 0, CUP_GRASP_DZ + .02], .045, 300)
    D.move(R, centre + [.0, -.0, CUP_GRASP_DZ + .14], .045, 600)
    D.move(R, PARK[R], .045, 1000)


def park(D):
    D.move(0, PARK[0], .045, 1200, both=(1, PARK[1], .045))


def best(mu, n=1, exclude=()):
    order = [i for i in np.argsort(mu)[::-1] if i not in exclude]
    return [int(i) for i in order[:n]]


def drive_plan(seed, target, params, liners, slots, cups, wet=False, hold_s=HOLD_S + 1.5):
    D = Driver(seed, target, params)
    park(D)
    for i, s in zip(liners, slots): place_liner(D, i, s, wet)
    for c, s in zip(cups, slots): place_cup(D, c, s)
    park(D)
    D.wait(int(hold_s / DT))
    return D


def reference(seed, target, params):
    p = params or parameters(seed); t = TGT[target]
    mu = p['mu_wet'] if t['wet'] else p['mu_dry']          # privileged: picks the liner(s) with the highest friction
    li = best(mu, t['cups'])
    slots = [1] if t['cups'] == 1 else [0, 1]
    return np.array(drive_plan(seed, target, p, li, slots, list(range(t['cups'])), wet=t['wet']).acts)

def drag(D, i=PROBE_LINER):
    # Right arm: grasp glass 1, hold it 10 mm above liner i (tare point), set it down and let go, re-grasp it where it rests
    # (removes the grasp-height offset), press 4 mm (commanded), drag it 30 mm along +y at a constant 10 mm/s, let go.
    # The card's signal: lateral wrist force / (glass weight + press). The drag window is recorded in _win.
    R = 1; x, y = CUP_XY[0]; cz = TABLE_Z + CUP_HH; lx = LINER_XY[i][0]
    D.move(R, [x, y, SAFE_Z], .045, 900)
    D.move(R, [x, y, cz + CUP_GRASP_DZ], .045, 700)
    D.move(R, None, 0., 350)
    D.move(R, [x, y, SAFE_Z], 0., 900)
    top = TABLE_Z + 2 * LINER_H[2] + CUP_HH + CUP_GRASP_DZ          # tool height with the glass resting on the liner
    D.move(R, [lx, DRAG_Y0, SAFE_Z], 0., 900)
    D.move(R, [lx, DRAG_Y0, top + .010], 0., 900)
    D.wait(300); _win['air'] = len(D.acts)
    D.move(R, [lx, DRAG_Y0, top + .002], 0., 600)
    D.move(R, None, .045, 300)
    D.move(R, [lx, DRAG_Y0, top], .045, 300)
    D.move(R, None, 0., 350)
    D.move(R, [lx, DRAG_Y0, top - PRESS], 0., 500, every=1)
    D.wait(300); _win['start'] = len(D.acts)
    D.move(R, [lx, DRAG_Y0 + DRAG_DY, top - PRESS], 0., int(DRAG_DY / DRAG_V / DT), linear=True, every=1)
    _win['end'] = len(D.acts)
    D.move(R, None, .045, 300)
    D.move(R, [lx, DRAG_Y0 + DRAG_DY, SAFE_Z], .045, 700)

def _probe_window(n_total, wet):
    # tick indices are plan-deterministic: recompute them once per probe kind
    key = 'wet' if wet else 'dry'
    if key not in _win:
        D = Driver(42, None, parameters(42)); park(D)
        if wet: dip_and_return(D, PROBE_LINER)
        drag(D); _win[key] = (_win['air'], _win['start'], _win['end'])
    return _win[key]


def mu_est(tr, wet):
    # Card signal: lateral wrist force / (glass weight + press). Right wrist, world frame, tared on the glass held 10 mm above
    # the liner: dFz = normal load on the liner, dFy = friction along the drag. Ratio of the means over the last 3/4 of the
    # constant-speed drag (the sliding contact chatters, so per-sample ratios are noisy; the mean impulse ratio is mu).
    a, s, e = _probe_window(len(tr['wrench']), wet)
    W = tr['wrench']; f0 = W[a - 200:a, 6:9].mean(0); seg = W[s + (e - s) // 4:e, 6:9] - f0
    return float(abs(seg[:, 1].mean()) / max(abs(seg[:, 2].mean()), .1))


def stat_mu(tr): return mu_est(tr, False)
def stat_mu_wet(tr): return mu_est(tr, True)

def return_cup_to_table(D, c=0):
    # Right arm: re-grasp a glass standing on the rack (top-down, jaw along X: the side faces at +-X stay vertical when the glass
    # tilts about X), carry it back to its start spot on the table, stand it there and let go.
    R = 1; g = D.d.xpos[D.k['cups'][c]].copy(); x, y = CUP_XY[c]; cz = TABLE_Z + CUP_HH
    D.move(R, [g[0], g[1], SAFE_Z], .045, 1000); D.move(R, g + [0, 0, CUP_GRASP_DZ], .045, 1000); D.move(R, None, 0., 350)
    D.move(R, [g[0], g[1], SAFE_Z], 0., 900)
    D.move(R, [x, y, SAFE_Z], 0., 1000)
    # the glass is still tilted 20 deg in the grip: end 15 mm above the upright set-down height so the rim does not jam into the
    # table (review round 2); it drops the last few mm and stands upright
    D.move(R, [x, y, cz + CUP_GRASP_DZ + .016], 0., 900); D.move(R, None, .045, 400)
    D.move(R, [x, y, SAFE_Z], .045, 700); D.move(R, PARK[R], .045, 900)


def _rt_fall(seed, positive):
    # positive: a glass released on a slipping liner slides off the rack (cup_fell). negative: a glass that held on the rack is
    # picked up again and stood back on the table (the card's recovery path): must not count as a fall.
    p = parameters(seed); mu = p['mu_dry']
    D = Driver(seed, None, p); park(D)
    i = int(np.argmin(mu)) if positive else best(mu, 1)[0]
    place_liner(D, i, 1); place_cup(D, 0, 1)
    if positive: park(D); D.wait(3000)
    else: D.wait(2000); return_cup_to_table(D, 0); D.wait(1500)
    return np.array(D.acts), p

              dict(name='cup_fell', event_key='cup_fell', positive=lambda seed: _rt_fall(seed, True), negative=lambda seed: _rt_fall(seed, False))]
